cmake_minimum_required(VERSION 3.16)
project(cerber_gps C)
set(CMAKE_C_STANDARD 11)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS ON)
find_package(PkgConfig REQUIRED)
pkg_check_modules(CJSON REQUIRED IMPORTED_TARGET libcjson)
pkg_check_modules(MOSQUITTO REQUIRED IMPORTED_TARGET libmosquitto)
find_package(Threads REQUIRED)
add_library(gps_parser STATIC src/gps_parser.c lib/src/minmea.c)
target_include_directories(gps_parser PUBLIC include PRIVATE lib/inc)
target_link_libraries(gps_parser PUBLIC PkgConfig::CJSON m)
add_library(gps_serial STATIC src/gps_serial.c)
target_include_directories(gps_serial PUBLIC include)
add_executable(robot_gps src/main.c)
target_link_libraries(robot_gps PRIVATE gps_parser gps_serial PkgConfig::MOSQUITTO Threads::Threads)
foreach(target gps_parser gps_serial robot_gps)
  target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic)
endforeach()
include(CTest)
if(BUILD_TESTING)
  add_executable(parser_test tests/parser_test.c)
  target_compile_options(parser_test PRIVATE -UNDEBUG)
  target_link_libraries(parser_test PRIVATE gps_parser)
  add_test(NAME gps_parser COMMAND parser_test)
  add_executable(serial_test tests/serial_test.c)
  target_compile_options(serial_test PRIVATE -UNDEBUG)
  target_link_libraries(serial_test PRIVATE gps_serial Threads::Threads)
  if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_link_libraries(serial_test PRIVATE util)
  endif()
  add_test(NAME gps_serial_pty COMMAND serial_test)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  add_executable(robot_gps_test_devices src/main.c)
  target_compile_definitions(robot_gps_test_devices PRIVATE GPS_TEST_DEVICES=1)
  target_compile_options(robot_gps_test_devices PRIVATE -Wall -Wextra -Wpedantic)
  target_link_libraries(robot_gps_test_devices PRIVATE gps_parser gps_serial PkgConfig::MOSQUITTO Threads::Threads)
  add_test(NAME gps_runtime_pty_mqtt
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/runtime_test.py
      $<TARGET_FILE:robot_gps_test_devices>)
  set_tests_properties(gps_runtime_pty_mqtt PROPERTIES TIMEOUT 30)
endif()
install(TARGETS robot_gps RUNTIME DESTINATION bin)
