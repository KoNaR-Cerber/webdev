cmake_minimum_required(VERSION 3.8)
project(cerber_telemetry)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
add_compile_options(-Wall -Wextra)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(tf2 REQUIRED)

add_library(telemetry_parser STATIC src/parser.cpp)
target_include_directories(telemetry_parser PUBLIC include)
add_executable(telemetry_receiver src/receiver.cpp)
target_link_libraries(telemetry_receiver telemetry_parser)
ament_target_dependencies(telemetry_receiver rclcpp sensor_msgs nav_msgs geometry_msgs std_msgs tf2)
install(TARGETS telemetry_receiver DESTINATION lib/${PROJECT_NAME})

if(BUILD_TESTING)
  enable_testing()
  add_executable(parser_test tests/parser_test.cpp)
  target_link_libraries(parser_test telemetry_parser)
  add_test(NAME wire_parser COMMAND parser_test)
endif()
ament_package()
